HiddenMonotonicity.TransformerDC.Transformer.dc_decomposition_linear_cost

Locally checked, unpublished Lean source. This is an addition to the pinned thesis formalization; it has not been committed or published as a GitHub revision.

The materialized shared evaluator computes both existing Transformer DC streams exactly, for arbitrary finite depth, heads, masks and widths. Its actual scalar-operation counter is bounded by 200 times the original shared scalar graph size. The same contract includes convexity, monotonicity, nonnegative streams and exact recovery on the doubled-input orthant, and convex exact heads on signed raw inputs. Arithmetic, comparisons and scalar special functions have unit cost; stored inputs and fixed parameters are available directly, and parameter-derived constants are precomputed. This is an arithmetic-work bound, not a bit-complexity, floating-point or compiled Lean wall-clock guarantee.

18/-- Each affine output has `n` scalar products, `n` accumulation additions,
19a bias source, and one bias addition. -/
20def affineSize (n m : ℕ) : ℕ := m * (2*n + 2)
21
22/-- A shared dot product has one multiplication and one accumulation per term. -/
23def dotSize (n : ℕ) : ℕ := 2*n
24
25/-- Exponentials, denominator accumulation, coordinate divisions, and one
26normalization source. The shared denominator is computed once for the row. -/
27def softmaxSize (n : ℕ) : ℕ := 3*n + 1
28
29/-- Mean and variance reductions, centering and squares, normalization, learned
30scale and bias: seven scalar operations per feature and five shared operations. -/
31def normSize (n : ℕ) : ℕ := 7*n + 5
32
33/-- Scalar-graph size of one attention head with cached Q, K and V projections. -/
34def headSize {t d : ℕ} (h : TransformerHead t d) : ℕ :=
35  t * (2 * affineSize d h.keyWidth + affineSize d h.valueWidth) +
36    ∑ q : Fin t,
37      (h.mask.keyCount q * (dotSize h.keyWidth + 1) +
38       softmaxSize (h.mask.keyCount q) +
39       h.valueWidth * dotSize (h.mask.keyCount q) +
40       d * dotSize h.valueWidth)
41
42/-- Both norms, cached attention heads, two residual merges, and the tokenwise
43MLP, with a scalar activation node for each hidden feature. -/
44def blockSize {t d : ℕ} (b : TransformerBlock t d) : ℕ :=
45  t * normSize d + (b.heads.map headSize).sum +
46    t*d*(b.heads.length + 3) + t * normSize d +
47    t*(affineSize d b.hiddenWidth + b.hiddenWidth + affineSize b.hiddenWidth d) + t*d
48
49/-- The shared graph adds the work of its blocks; earlier blocks are never
50expanded again at later references. -/
51def blocksSize {t d : ℕ} : List (TransformerBlock t d) → ℕ
52  | [] => 0
53  | b :: bs => blockSize b + blocksSize bs
54
55/-- Nested feature/token reductions for each signed affine output. -/
56def readoutSize (t d o : ℕ) : ℕ := o*(2*(t*d) + t + 2)
57
58/-- Positional bias, arbitrary finite block depth, optional final norm, and all
59output logits. This size counts the original shared graph, not an expression tree. -/
60def networkSize {t d o : ℕ} (net : Transformer t d o) : ℕ :=
61  2*(t*d) + blocksSize net.blocks +
62    (match net.finalNorm with | none => 0 | some _ => t * normSize d) +
63    readoutSize t d o
64
65
66/-- Exact affine pair schedule cost: scaling, accumulation, bias split, and add. -/
67def affineWork (n m : ℕ) : ℕ := m*(6*n+5)
68def softmaxWork (n : ℕ) : ℕ := 8*n+4
69def normWork (n : ℕ) : ℕ := 39*n+18
70
71def headWork {t d : ℕ} (h : TransformerHead t d) : ℕ :=
72  t * (2 * affineWork d h.keyWidth + affineWork d h.valueWidth) +
73    ∑ q : Fin t,
74      (h.mask.keyCount q * (14*h.keyWidth + 4) +
75       softmaxWork (h.mask.keyCount q) +
76       h.valueWidth * (14*h.mask.keyCount q) + d * (6*h.valueWidth))
77
78/-- All terms count the schedule directly; 32 uniformly bounds each activation. -/
79def blockWork {t d : ℕ} (b : TransformerBlock t d) : ℕ :=
80  t * normWork d + (b.heads.map headWork).sum +
81    t*d*(2*b.heads.length + 7) + t * normWork d +
82    t*(affineWork d b.hiddenWidth + 32*b.hiddenWidth + affineWork b.hiddenWidth d) + 2*(t*d)
83
84def blocksWork {t d : ℕ} : List (TransformerBlock t d) → ℕ
85  | [] => 0
86  | b :: bs => blockWork b + blocksWork bs
87
88def readoutWork (t d o : ℕ) : ℕ := o*(6*(t*d)+2*t+5)
89
90def networkWork {t d o : ℕ} (net : Transformer t d o) : ℕ :=
91  5*(t*d) + blocksWork net.blocks +
92    (match net.finalNorm with | none => 0 | some _ => t * normWork d) +
93    readoutWork t d o

107lemma headWork_le {t d : ℕ} (h : TransformerHead t d) : headWork h ≤ 200 * headSize h := by
108  have hq : (∑ q : Fin t,
109      (h.mask.keyCount q * (14*h.keyWidth + 4) +
110       softmaxWork (h.mask.keyCount q) +
111       h.valueWidth * (14*h.mask.keyCount q) + d * (6*h.valueWidth))) ≤
112      ∑ q : Fin t, 200 *
113      (h.mask.keyCount q * (dotSize h.keyWidth + 1) +
114       softmaxSize (h.mask.keyCount q) +
115       h.valueWidth * dotSize (h.mask.keyCount q) + d * dotSize h.valueWidth) := by
116    apply Finset.sum_le_sum
117    intro q _
118    dsimp [dotSize, softmaxWork, softmaxSize]
119    nlinarith
120  rw [← Finset.mul_sum] at hq
121  have hp := Nat.mul_le_mul_left t
122    (Nat.add_le_add (Nat.mul_le_mul_left 2 (affineWork_le d h.keyWidth))
123      (affineWork_le d h.valueWidth))
124  dsimp [headWork, headSize]
125  nlinarith

136lemma blockWork_le {t d : ℕ} (b : TransformerBlock t d) :
137    blockWork b ≤ 200 * blockSize b := by
138  have hn := Nat.mul_le_mul_left t (normWork_le d)
139  have hh := list_heads_work_le b.heads
140  have hup := Nat.mul_le_mul_left t (affineWork_le d b.hiddenWidth)
141  have hdown := Nat.mul_le_mul_left t (affineWork_le b.hiddenWidth d)
142  have hres : t*d*(2*b.heads.length+7) ≤ 200*(t*d*(b.heads.length+3)) := by
143    calc
144      t*d*(2*b.heads.length+7) ≤ t*d*(200*(b.heads.length+3)) :=
145        Nat.mul_le_mul_left _ (by omega)
146      _ = _ := by ring
147  have hact : 32*(t*b.hiddenWidth) ≤ 200*(t*b.hiddenWidth) := by omega
148  have hlast : 2*(t*d) ≤ 200*(t*d) := by omega
149  dsimp [blockWork, blockSize]
150  nlinarith only [hn, hh, hup, hdown, hres, hact, hlast]

161lemma readoutWork_le (t d o : ℕ) : readoutWork t d o ≤ 200 * readoutSize t d o := by
162  dsimp [readoutWork, readoutSize]
163  calc
164    o*(6*(t*d)+2*t+5) ≤ o*(200*(2*(t*d)+t+2)) :=

168lemma networkWork_le {t d o : ℕ} (net : Transformer t d o) :
169    networkWork net ≤ 200 * networkSize net := by
170  have hb := blocksWork_le net.blocks
171  have hr := readoutWork_le t d o
172  have hn := Nat.mul_le_mul_left t (normWork_le d)

239/-- The exact same globally defined DC streams admit a shared evaluation with
240uniform linear arithmetic cost. The cost statement is proved for the materialized
241writer computation, not for the earlier recursive expression-tree evaluator. -/
242theorem Transformer.dc_decomposition_linear_cost {t d o : ℕ} (net : Transformer t d o) :
243    (∀ x i, (net.sharedDecompose x).val.get i = net.decompose x i) ∧
244    (∀ x, (net.sharedDecompose x).steps ≤ 200 * networkSize net) ∧
245    MonotoneOn net.decompose (Set.Ici (0 : DoubledInput (Fin t × Fin d))) ∧
246    (∀ i, ConvexOn ℝ (Set.Ici (0 : DoubledInput (Fin t × Fin d)))
247      (fun x => (net.decompose x i).1)) ∧
248    (∀ i, ConvexOn ℝ (Set.Ici (0 : DoubledInput (Fin t × Fin d)))
249      (fun x => (net.decompose x i).2)) ∧
250    (∀ x, 0 ≤ x → 0 ≤ net.decompose x) ∧
251    (∀ x i, (net.decompose x i).1 - (net.decompose x i).2 =
252      net.eval (x.1 - x.2) i) ∧
253    (∀ i, ConvexOn ℝ Set.univ (fun x : Fin t × Fin d → ℝ =>
254      (net.rawDecompose x i).1)) ∧
255    (∀ i, ConvexOn ℝ Set.univ (fun x : Fin t × Fin d → ℝ =>
256      (net.rawDecompose x i).2)) ∧
257    (∀ x, 0 ≤ net.rawDecompose x) ∧
258    (∀ x i, (net.rawDecompose x i).1 - (net.rawDecompose x i).2 = net.eval x i) := by
259  obtain ⟨hm,hp,hq,hn,he⟩ := net.dc_decomposition
260  obtain ⟨hrp,hrq,hrn,hre⟩ := net.raw_dc_decomposition
261  exact ⟨net.sharedDecompose_eq,net.sharedDecompose_steps_le,hm,hp,hq,hn,he,hrp,hrq,hrn,hre⟩
